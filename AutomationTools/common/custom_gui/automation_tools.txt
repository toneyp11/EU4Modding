##############################################################################
# AutomationTools - UI LAYER (custom GUI bindings / input)
# ---------------------------------------------------------------------------
# Binds the scripted GUI elements in interface/provinceview.gui to the
# UI<->backend contract. Each object's `name` MUST match a `scripted = yes`
# element in the .gui file. This layer ONLY reads/writes contract flags and hub
# variables (see on_actions header) - it contains NO game logic and never calls
# backend effects, so redesigning the panel means editing only this file, the
# .gui, and localisation. The backend never looks at these names.
##############################################################################

# === Panel framework: open / close ==========================================

# Master button on the province view - toggles the panel's visibility flag.
custom_button = {
	name = atools_master_toggle
	potential = { always = yes }
	trigger = { always = yes }
	effect = {
		if = { limit = { has_global_flag = atools_panel_open } clr_global_flag = atools_panel_open }
		else = { set_global_flag = atools_panel_open }
	}
	tooltip = AUTOMATION_TOOLS_BTN_TT
}

# The panel window - visible only while the flag is set (gates all children).
custom_window = {
	name = atools_panel
	potential = { has_global_flag = atools_panel_open }
}

# Close (X) button inside the panel.
custom_button = {
	name = atools_close
	potential = { always = yes }
	trigger = { always = yes }
	effect = { clr_global_flag = atools_panel_open }
	tooltip = AUTOMATION_CLOSE_TT
}

# === Tool controls: Auto-cede from #1 nation ================================

# On/off toggle -> contract flag atools_cede_enabled (frame 2 = on, 1 = off).
custom_button = {
	name = atools_toggle_cede
	potential = { always = yes }
	trigger = { always = yes }
	effect = {
		if = { limit = { has_global_flag = atools_cede_enabled } clr_global_flag = atools_cede_enabled }
		else = { set_global_flag = atools_cede_enabled }
	}
	frame = { number = 2 trigger = { has_global_flag = atools_cede_enabled } }
	frame = { number = 1 trigger = { always = yes } }
	tooltip = ATOOLS_CEDE_TT
}

# Interval selector -> contract var atools_cede_interval. One button cycling
# 1 -> 6 -> 12 -> 24 months (value shown in the status text). check_variable is >=.
custom_button = {
	name = atools_interval_cycle
	potential = { always = yes }
	trigger = { always = yes }
	effect = {
		1 = {
			if = {
				limit = { check_variable = { which = atools_cede_interval value = 1 } NOT = { check_variable = { which = atools_cede_interval value = 6 } } }
				set_variable = { which = atools_cede_interval value = 6 }
			}
			else_if = {
				limit = { check_variable = { which = atools_cede_interval value = 6 } NOT = { check_variable = { which = atools_cede_interval value = 12 } } }
				set_variable = { which = atools_cede_interval value = 12 }
			}
			else_if = {
				limit = { check_variable = { which = atools_cede_interval value = 12 } NOT = { check_variable = { which = atools_cede_interval value = 24 } } }
				set_variable = { which = atools_cede_interval value = 24 }
			}
			else = { set_variable = { which = atools_cede_interval value = 1 } }
		}
	}
	tooltip = ATOOLS_INTERVAL_CYCLE_TT
}

# Read-out: current interval + total cessions (text keyed by name in localisation).
custom_text_box = {
	name = atools_cede_status
	potential = { always = yes }
	tooltip = ATOOLS_CEDE_STATUS_TT
}
