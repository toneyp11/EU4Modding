##############################################################################
# AutomationTools - BACKEND (framework: startup / state init)
# ---------------------------------------------------------------------------
# Initialises the hub state and starts the global timer, once per game.
#
# ============================ UI <-> BACKEND CONTRACT =======================
# The UI and backend communicate ONLY through these. Keep them stable and the
# UI can be redesigned without touching any backend file (and vice-versa).
#
# GLOBAL FLAGS
#   atools_panel_open       UI-only   : tools panel is open (window visibility)
#   atools_cede_enabled     UI writes / backend reads : anti-snowball tool on/off
#                           (release possible nations first, else cede to weakest)
#   atools_convert_enabled  UI writes / backend reads : auto-convert religion+culture on/off
#
# HUB VARIABLES (on province 1; UI displays via [1.<var>.GetValue])
#   atools_cede_interval    UI writes / backend reads : months between shrink runs (>=1)
#   atools_cede_count       backend writes / UI reads : total shrinks this game
#   atools_convert_interval UI writes / backend reads : months between convert runs (>=1)
#
# BACKEND-INTERNAL (not part of the UI contract; do not touch from UI)
#   atools_months           shrink timer's elapsed-month counter (hub var)
#   atools_convert_months   convert timer's elapsed-month counter (hub var)
#   atools_timer_started    guard so the timer chain starts only once
#   atools_have_top / atools_have_receiver : per-run guards inside the cede effect
#   atools_td_before / atools_td_after : top nation's dev snapshot (release detection)
#   event targets: atools_top_nation, atools_receiver
# ===========================================================================
##############################################################################

# Start the monthly timer exactly once. on_startup fires on new games AND every
# save load, so the flag guard prevents duplicate timer chains (pending delayed
# events persist in the save, so the chain itself survives reloads).
on_startup = {
	if = {
		limit = { NOT = { has_global_flag = atools_timer_started } }
		set_global_flag = atools_timer_started
		1 = {
			set_variable = { which = atools_months value = 0 }
			set_variable = { which = atools_cede_interval value = 12 }      # default: shrink every 12 months
			set_variable = { which = atools_cede_count value = 0 }
			set_variable = { which = atools_convert_months value = 0 }
			set_variable = { which = atools_convert_interval value = 60 }   # default: convert every 60 months (5 yr) - slow drift
			province_event = { id = atools.1 days = 30 }
		}
	}
}
