##############################################################################
# AutomationTools - BACKEND (tool logic)
# ---------------------------------------------------------------------------
# Depends ONLY on the UI<->backend contract (global flags + province-1 "hub"
# variables). It never references GUI element names, so the UI can be redesigned
# freely without touching this file. Contract is documented in
# docs/automationtools.md and common/on_actions/atools_on_actions.txt.
#
# Reliability notes (hard-won):
#  - `exists = event_target:X` is unreliable here (returns false even when the
#    target IS saved), so we guard with GLOBAL FLAGS instead.
#  - Biggest/weakest use an exact running max/min over every_country /
#    every_neighbor_country (the vanilla flavorBYZ idiom). `= PREV` never resolves.
#  - NEVER put a scope command (.GetName/.GetValue) in a log -> stack-overflow crash.
#
# DEFENSIVE DESIGN (added while hunting the ~1590 STACK_OVERFLOW, which is an
# infinite recursion in EU4 name/string building - see docs/eu4-modding-reference.md):
# every step is guarded so the tools can never hand the engine a degenerate state -
# no nation is emptied, no capital is moved, no colonial nation is given vassals,
# marker flags cannot go stale, cores are not stacked without bound, and no
# country-scope variables are left behind (EU4 has no clear_variable, so anything
# written there persists in the save forever).
##############################################################################

# --- Shared step: cede ONE border province to the weakest cede-able neighbour --
# Called in the #1 nation's scope, either as the fallback when nothing was
# releasable or as the only shrink route for a colonial #1.
atools_shrink_cede_effect = {
	log = "ATOOLS shrink F: fallback-cede"
	clr_global_flag = atools_allow_capital_cede

	# A receiver only counts if we can ACTUALLY cede to it. That test must match the
	# province filter below exactly: the neighbour has to border a NON-CAPITAL city of
	# the #1. Testing "any city" while the cede refuses capitals let a small nation
	# (e.g. Granada, whose only province touching its weakest neighbour was its own
	# capital) pick a receiver it could never hand anything to - stuck every month.
	#
	# Pass 1: prefer a NON-colonial receiver. Handing provinces to a colonial nation
	# can re-parent colonial nations, and a loop in that parent chain is the leading
	# suspect for the name-resolution crash.
	every_neighbor_country = {
		if = {
			limit = {
				is_colonial_nation = no
				any_owned_province = { any_neighbor_province = { is_city = yes is_capital = no owned_by = event_target:atools_top_nation } }
				OR = {
					NOT = { has_global_flag = atools_have_receiver }
					NOT = { total_development = event_target:atools_receiver }
				}
			}
			save_event_target_as = atools_receiver
			set_global_flag = atools_have_receiver
		}
	}
	# Pass 2: no non-colonial candidate (e.g. a colonial #1 in the Americas whose
	# neighbours are all colonial) -> allow colonial receivers rather than stall.
	if = {
		limit = { NOT = { has_global_flag = atools_have_receiver } }
		every_neighbor_country = {
			if = {
				limit = {
					any_owned_province = { any_neighbor_province = { is_city = yes is_capital = no owned_by = event_target:atools_top_nation } }
					OR = {
						NOT = { has_global_flag = atools_have_receiver }
						NOT = { total_development = event_target:atools_receiver }
					}
				}
				save_event_target_as = atools_receiver
				set_global_flag = atools_have_receiver
			}
		}
	}
	# Pass 3 (last resort): the #1 has no non-capital province touching anyone - the
	# only way to shrink it is to give up the capital. EU4 relocates the seat of
	# government automatically. Better than being wedged on the same nation forever.
	if = {
		limit = { NOT = { has_global_flag = atools_have_receiver } }
		set_global_flag = atools_allow_capital_cede
		every_neighbor_country = {
			if = {
				limit = {
					any_owned_province = { any_neighbor_province = { is_city = yes owned_by = event_target:atools_top_nation } }
					OR = {
						NOT = { has_global_flag = atools_have_receiver }
						NOT = { total_development = event_target:atools_receiver }
					}
				}
				save_event_target_as = atools_receiver
				set_global_flag = atools_have_receiver
			}
		}
	}
	log = "ATOOLS shrink G: min-done"

	if = {
		limit = { NOT = { has_global_flag = atools_have_receiver } }
		log = "ATOOLS shrink G2: NO-RECEIVER (no bordering country at all)"
	}
	if = {
		limit = { has_global_flag = atools_have_receiver }
		random_owned_province = {
			limit = {
				is_city = yes
				# Normally never cede a capital (it forces the engine to relocate the seat
				# of government). Pass 3 above sets the flag when the capital is the ONLY
				# thing that can be given up, so we prefer that over stalling forever.
				OR = {
					is_capital = no
					has_global_flag = atools_allow_capital_cede
				}
				any_neighbor_province = { owned_by = event_target:atools_receiver }
			}
			log = "ATOOLS shrink H: province-picked"
			# Only add a core if the receiver lacks one. Adding unconditionally stacked
			# cores on the same provinces every run for the whole game, which also
			# inflates what release_all_possible_countries can later release.
			if = {
				limit = { NOT = { is_core = event_target:atools_receiver } }
				add_core = event_target:atools_receiver
			}
			cede_province = event_target:atools_receiver
			log = "ATOOLS shrink I: province-ceded"
			# Only strip the giver's core if it actually had one - avoids "Failed to
			# remove prov N as core, it hasnt got it" error spam.
			if = {
				limit = { is_core = event_target:atools_top_nation }
				remove_core = event_target:atools_top_nation
			}
			1 = { change_variable = { which = atools_cede_count value = 1 } }
			set_global_flag = atools_shrink_done      # this attempt actually shrank someone
		}
	}
	log = "ATOOLS shrink J: fallback-done"
}

# --- One shrink ATTEMPT: pick the biggest not-yet-skipped nation and try to shrink it.
# Sets atools_shrink_done on success; on failure it flags that nation so the next attempt
# moves on. EU4 has NO "can this nation be released?" trigger (has_releasable_nation /
# can_release_nation / num_of_released_nations do not exist), so the only way to know
# whether a nation can be shrunk is to try.
atools_shrink_attempt_effect = {
	clr_global_flag = atools_have_top
	clr_global_flag = atools_have_receiver

	# Top nation = the EXACT most-developed country (running maximum; first candidate
	# seeds it, then anything >= replaces it). Subjects and colonial nations are eligible.
	# Requires >= 2 cities so a cede can never wipe a nation off the map.
	# NOTE: deliberately NO "must have a neighbour" filter. An island nation (Britain,
	# Japan) has no land or strait neighbour but can still be shrunk by RELEASING nations
	# from inside itself; filtering on neighbours skipped those before they got the chance.
	every_country = {
		if = {
			limit = {
				num_of_cities = 2
				NOT = { has_country_flag = atools_skip_top }
				OR = {
					NOT = { has_global_flag = atools_have_top }
					total_development = event_target:atools_top_nation
				}
			}
			save_event_target_as = atools_top_nation
			set_global_flag = atools_have_top
		}
	}

	if = {
		limit = { has_global_flag = atools_have_top }
		log = "ATOOLS shrink A: top-found"
		event_target:atools_top_nation = {
			# DIAGNOSTIC: stamp the current #1 so a save captured before a crash names the
			# nation we last acted on (.GetName in a log is itself a crash, so we cannot log it).
			set_country_flag = atools_last_top

			if = {
				# A colonial nation shrinks by CEDING only. Releasing from one would make it the
				# overlord of its own vassals while it is itself a subject of its colonial parent.
				limit = { is_colonial_nation = yes }
				log = "ATOOLS shrink K: colonial-top-cede-only"
				atools_shrink_cede_effect = yes
			}
			else = {
				# PRIMARY: release everything releasable, then free the NEW subjects only.
				# This needs no neighbour, so it works for island nations.
				log = "ATOOLS shrink B: pre-release_all_possible"
				# Clear THEN set: a country dead at global pre-clean time cannot be reached by
				# every_country, so a revived tag can return carrying a stale flag from a past life.
				every_subject_country = { clr_country_flag = atools_preexisting_sub }
				every_subject_country = { set_country_flag = atools_preexisting_sub }
				release_all_possible_countries = yes
				log = "ATOOLS shrink C: post-release_all_possible"

				if = {
					# Did the release create anything? Ask the subject list directly rather than
					# snapshotting development into country variables (EU4 has no clear_variable,
					# so those persist in the save forever).
					limit = { any_subject_country = { NOT = { has_country_flag = atools_preexisting_sub } } }
					every_subject_country = {
						limit = {
							NOT = { has_country_flag = atools_preexisting_sub }
							is_colonial_nation = no
						}
						grant_independence = yes
					}
					log = "ATOOLS shrink E: released-ok"
					1 = { change_variable = { which = atools_cede_count value = 1 } }
					set_global_flag = atools_shrink_done
				}
				else = { atools_shrink_cede_effect = yes }
			}
		}

		# Nothing worked on this nation - nothing releasable AND nobody to cede to (a fully
		# cored island, or a colonial nation ringed by uncolonized land). Flag it so the next
		# attempt targets the next-largest instead of wedging here forever.
		if = {
			limit = { NOT = { has_global_flag = atools_shrink_done } }
			log = "ATOOLS shrink S: un-shrinkable, skipping to next"
			event_target:atools_top_nation = { set_country_flag = atools_skip_top }
		}
	}
}

# === Tool: Auto-shrink the #1 nation (anti-snowball) ========================
# Each run the most-developed nation is shrunk: it first releases every nation it can as
# INDEPENDENT states; if nothing was releasable it cedes one border province to its
# weakest neighbour. A nation that can do neither is skipped and the next-largest is
# tried (EU4 has no loops, so the retry is unrolled). Called from on_monthly_pulse.
atools_cede_effect = {
	clr_global_flag = atools_shrink_done

	# Per-run scratch flags - clear globally first (see the revived-tag note above).
	every_country = { limit = { has_country_flag = atools_preexisting_sub } clr_country_flag = atools_preexisting_sub }
	every_country = { limit = { has_country_flag = atools_last_top } clr_country_flag = atools_last_top }
	every_country = { limit = { has_country_flag = atools_skip_top } clr_country_flag = atools_skip_top }

	atools_shrink_attempt_effect = yes
	if = { limit = { NOT = { has_global_flag = atools_shrink_done } } atools_shrink_attempt_effect = yes }
	if = { limit = { NOT = { has_global_flag = atools_shrink_done } } atools_shrink_attempt_effect = yes }

	# Post-clean so nothing survives this run.
	every_country = { limit = { has_country_flag = atools_preexisting_sub } clr_country_flag = atools_preexisting_sub }
	every_country = { limit = { has_country_flag = atools_skip_top } clr_country_flag = atools_skip_top }
}

# === Tool: Auto-convert religion & culture ==================================
# Fights religious/cultural stagnation (vanilla AI almost never converts same-group
# "heretic" provinces or foreign cultures). Each run, every owned province whose
# religion/culture differs from its owner's has a CHANCE to flip toward the owner,
# so the map homogenises gradually rather than all at once. change_religion /
# change_culture accept a country scope; `owner` = the province's owner.
atools_convert_effect = {
	# TEMP traces: the convert pass had NO logging, so a crash during/just after a
	# conversion sweep was indistinguishable from one during the shrink. change_culture
	# in particular makes the engine re-derive province names (they are culture-specific),
	# which is string building - the exact area the stack overflow lives in.
	log = "ATOOLS convert: start"
	every_province = {
		limit = {
			is_city = yes
			owner = { always = yes }              # defensive: must have a valid owner scope
			controlled_by = owner                 # skip occupied / besieged provinces
		}
		if = {
			limit = { NOT = { religion = owner } }
			random = { chance = 20 change_religion = owner }
		}
		if = {
			limit = { NOT = { culture = owner } }
			random = { chance = 20 change_culture = owner }
		}
	}
	log = "ATOOLS convert: done"
}
