##############################################################################
# AutomationTools - BACKEND (tool logic)
# ---------------------------------------------------------------------------
# Depends ONLY on the UI<->backend contract (global flags + province-1 "hub"
# variables). It never references GUI element names, so the UI can be redesigned
# freely without touching this file. Contract is documented in
# docs/automationtools.md and common/on_actions/atools_on_actions.txt.
#
# Reliability notes (hard-won):
#  - `exists = event_target:X` is unreliable here (returns false even when the
#    target IS saved), so we guard with GLOBAL FLAGS instead.
#  - Finding the biggest/weakest nation uses vanilla's threshold-ladder idiom
#    (numeric total_development + any_country/random_country), not mid-iteration
#    event_target comparisons (which don't resolve).
##############################################################################

# === Tool: Auto-cede from #1 nation (anti-snowball) =========================
# The highest-development nation cedes one border province to a weak neighbour.
# Called from the atools.1 timer when the tool is enabled. Selection is by
# development BRACKET (tunable below), not exact rank.
atools_cede_effect = {
	clr_global_flag = atools_have_top
	clr_global_flag = atools_have_receiver

	# 1) Top nation = the EXACT most-developed country (running maximum). Subjects eligible
	# (including colonial nations - they can be shrunk fine).
	every_country = {
		if = {
			limit = {
				OR = {
					NOT = { has_global_flag = atools_have_top }
					total_development = event_target:atools_top_nation
				}
			}
			save_event_target_as = atools_top_nation
			set_global_flag = atools_have_top
		}
	}

	if = {
		limit = { has_global_flag = atools_have_top }
		log = "ATOOLS shrink A: top-found"
		# DIAGNOSTIC: stamp the current #1 with a country flag (cleared from the previous
		# one) so a save captured just before a crash names the nation we last acted on.
		# Country flags persist in the save; we cannot log names (.GetName in a log crashes).
		every_country = { limit = { has_country_flag = atools_last_top } clr_country_flag = atools_last_top }
		event_target:atools_top_nation = { set_country_flag = atools_last_top }
		event_target:atools_top_nation = {

			# PRIMARY: release every releasable nation as INDEPENDENT, safely.
			# Freeing the #1's PRE-EXISTING subjects (colonial nations / subjects tangled in
			# alliances-PUs-coalitions) is what overflows vanilla's post-processing in the
			# late game. So: mark those first, release the fresh nations, then free ONLY the
			# fresh ones (static names, no ties -> shallow, safe), leaving complex subjects be.
			export_to_variable = { which = atools_td_before value = total_development }
			log = "ATOOLS shrink B: pre-release_all_possible"
			every_subject_country = { set_country_flag = atools_preexisting_sub }
			release_all_possible_countries = yes
			log = "ATOOLS shrink C: post-release_all_possible"
			every_subject_country = {
				limit = { NOT = { has_country_flag = atools_preexisting_sub } }
				grant_independence = yes
			}
			log = "ATOOLS shrink D: post-free-new-only"
			every_subject_country = { clr_country_flag = atools_preexisting_sub }
			export_to_variable = { which = atools_td_after value = total_development }

			if = {
				limit = { NOT = { check_variable = { which = atools_td_after which = atools_td_before } } }
				log = "ATOOLS shrink E: released-ok"
				1 = { change_variable = { which = atools_cede_count value = 1 } }
			}
			else = {
				# FALLBACK: nothing releasable -> cede a border province to the weakest neighbour.
				log = "ATOOLS shrink F: fallback-cede"
				every_neighbor_country = {
					if = {
						limit = {
							# Only neighbours the #1 can actually CEDE to: this neighbour must
							# own a province bordering one of the #1's cities. Without this, the
							# weakest neighbour by dev may share no city border, so the cede below
							# finds no province and the #1 never shrinks (the "stuck" bug).
							any_owned_province = { any_neighbor_province = { is_city = yes owned_by = event_target:atools_top_nation } }
							OR = {
								NOT = { has_global_flag = atools_have_receiver }
								NOT = { total_development = event_target:atools_receiver }
							}
						}
						save_event_target_as = atools_receiver
						set_global_flag = atools_have_receiver
					}
				}
				log = "ATOOLS shrink G: min-done"
				if = {
					limit = { has_global_flag = atools_have_receiver }
					random_owned_province = {
						limit = {
							is_city = yes
							any_neighbor_province = { owned_by = event_target:atools_receiver }
						}
						log = "ATOOLS shrink H: province-picked"
						add_core = event_target:atools_receiver
						cede_province = event_target:atools_receiver
						log = "ATOOLS shrink I: province-ceded"
						if = { limit = { is_core = event_target:atools_top_nation } remove_core = event_target:atools_top_nation }
						1 = { change_variable = { which = atools_cede_count value = 1 } }
					}
				}
				log = "ATOOLS shrink J: fallback-done"
			}
		}
	}
}

# === Tool: Auto-convert religion & culture ==================================
# Fights religious/cultural stagnation (vanilla AI almost never converts same-group
# "heretic" provinces or foreign cultures). Each interval, every owned province whose
# religion/culture differs from its owner's has a CHANCE to flip toward the owner, so
# the map homogenises gradually rather than all at once. change_religion/change_culture
# accept a country scope; `owner` = the province's owner (religion + primary culture).
# The chance below is the per-province, per-interval percent (tune to taste).
atools_convert_effect = {
	every_province = {
		limit = { is_city = yes }
		if = {
			limit = { NOT = { religion = owner } }
			random = { chance = 20 change_religion = owner }
		}
		if = {
			limit = { NOT = { culture = owner } }
			random = { chance = 20 change_culture = owner }
		}
	}
}
