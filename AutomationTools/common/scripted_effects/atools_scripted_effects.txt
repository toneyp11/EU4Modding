# Automation Tools - scripted effects
# (INSTRUMENTED with log = "ATOOLS: ..." -> game.log while debugging. Removed once working.)
#
# NOTE: we do NOT use PREV for the comparisons - `total_development = PREV` does not
# resolve (vanilla never uses it and it produced an invalid scope). Instead we save
# each candidate as event_target:atools_candidate and compare against that, which is
# the verified-working idiom (tag = event_target:X, total_development = <scope>).

atools_cede_effect = {
	log = "ATOOLS: ---- cede effect running ----"
	clear_global_event_target = atools_top_nation
	clear_global_event_target = atools_receiver
	clear_global_event_target = atools_candidate

	# 1) Largest INDEPENDENT nation by total development.
	#    A candidate is #1 if no OTHER country has development >= its own.
	every_country = {
		limit = { is_subject = no }
		save_event_target_as = atools_candidate
		if = {
			limit = {
				NOT = {
					any_country = {
						NOT = { tag = event_target:atools_candidate }
						total_development = event_target:atools_candidate
					}
				}
			}
			save_global_event_target_as = atools_top_nation
		}
	}

	if = {
		limit = { exists = event_target:atools_top_nation }
		event_target:atools_top_nation = { log = "ATOOLS: top nation = [This.GetName]" }

		# DIAGNOSTIC (logs once): list every independent neighbour of the top nation.
		# (Only scope NAME commands work in logs - variable .GetValue is unsupported
		#  and stack-overflows, so we log names, not dev numbers.)
		if = {
			limit = { NOT = { has_global_flag = atools_dbg_logged } }
			set_global_flag = atools_dbg_logged
			log = "ATOOLS DBG: ---- independent neighbours of top nation ----"
			every_country = {
				limit = {
					is_subject = no
					NOT = { tag = event_target:atools_top_nation }
					is_neighbor_of = event_target:atools_top_nation
				}
				log = "ATOOLS DBG neighbour: [This.GetName]"
			}
			log = "ATOOLS DBG: ---- end neighbours ----"
		}

		# 2) WEAKEST independent nation bordering the top nation.
		#    Candidate qualifies if no OTHER bordering nation is strictly weaker.
		every_country = {
			limit = {
				is_subject = no
				NOT = { tag = event_target:atools_top_nation }
				is_neighbor_of = event_target:atools_top_nation
			}
			save_event_target_as = atools_candidate
			if = {
				limit = {
					NOT = {
						any_country = {
							is_subject = no
							NOT = { tag = event_target:atools_top_nation }
							NOT = { tag = event_target:atools_candidate }
							is_neighbor_of = event_target:atools_top_nation
							NOT = { total_development = event_target:atools_candidate }  # strictly weaker exists
						}
					}
				}
				save_global_event_target_as = atools_receiver
			}
		}

		if = {
			limit = { exists = event_target:atools_receiver }
			event_target:atools_receiver = { log = "ATOOLS: receiver = [This.GetName]" }

			# 3) Cede a border province from top nation to receiver.
			event_target:atools_top_nation = {
				random_owned_province = {
					limit = {
						is_city = yes
						any_neighbor_province = { owned_by = event_target:atools_receiver }
					}
					add_core = event_target:atools_receiver
					cede_province = event_target:atools_receiver
					remove_core = event_target:atools_top_nation
					1 = { change_variable = { which = atools_cede_count value = 1 } }
					log = "ATOOLS: CEDED province [This.GetName]"
				}
			}
		}
		else = { log = "ATOOLS: no receiver (top has no independent neighbour)" }
	}
	else = { log = "ATOOLS: no top nation found" }
}
