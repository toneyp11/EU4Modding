# AutomationTools - BACKEND (tool logic)
# ---------------------------------------------------------------------------
# Depends ONLY on the UI<->backend contract (global flags + province-1 "hub"
# variables). It never references GUI element names, so the UI can be redesigned
# freely without touching this file. Contract is documented in
# docs/automationtools.md and common/on_actions/atools_on_actions.txt.
#
# Reliability notes (hard-won):
#  - `exists = event_target:X` is unreliable here (returns false even when the
#    target IS saved), so we guard with GLOBAL FLAGS instead.
#  - Finding the biggest/weakest nation uses vanilla's threshold-ladder idiom
#    (numeric total_development + any_country/random_country), not mid-iteration
#    event_target comparisons (which don't resolve).

# === Tool: Auto-cede from #1 nation (anti-snowball) =========================
# The highest-development nation cedes one border province to a weak neighbour.
# Called from the atools.1 timer when the tool is enabled. Selection is by
# development BRACKET (tunable below), not exact rank.
atools_cede_effect = {
	clr_global_flag = atools_have_top
	clr_global_flag = atools_have_receiver

	# 1) Top nation = the EXACT most-developed country (running maximum, the vanilla
	# flavorBYZ idiom). Each candidate whose dev >= the current best overwrites it; the
	# first candidate seeds it (no flag yet). Ties -> last one iterated wins.
	# NOTE: subjects ARE eligible - a huge vassal can be the #1, else the most-developed
	# nation in the world would escape anti-snowball just by being a subject.
	every_country = {
		if = {
			limit = {
				OR = {
					NOT = { has_global_flag = atools_have_top }
					total_development = event_target:atools_top_nation
				}
			}
			save_event_target_as = atools_top_nation
			set_global_flag = atools_have_top
		}
	}

	if = {
		limit = { has_global_flag = atools_have_top }
		event_target:atools_top_nation = {

			# 2) PRIMARY shrink: carve out every nation releasable from its owned land,
			# then free them (and any pre-existing subjects) so they end up INDEPENDENT.
			# release_all_possible_countries releases as vassals; release_all_subjects
			# then frees them. Snapshot total development around it: if it fell, land
			# left the nation => a release actually happened.
			export_to_variable = { which = atools_td_before value = total_development }
			release_all_possible_countries = yes
			release_all_subjects = yes
			export_to_variable = { which = atools_td_after value = total_development }

			if = {
				# Development dropped -> the primary mechanism did the shrinking. Done.
				limit = { NOT = { check_variable = { which = atools_td_after which = atools_td_before } } }
				1 = { change_variable = { which = atools_cede_count value = 1 } }
			}
			else = {
				# 3) FALLBACK: nothing was releasable -> cede a border province to the
				# EXACT weakest neighbour (running minimum: a candidate strictly below the
				# current best overwrites it; first candidate seeds it). Ties -> first one
				# kept. Subjects ARE eligible recipients (own or another's vassal); a
				# country is never its own neighbour, so it can't pick itself.
				every_neighbor_country = {
					if = {
						limit = {
							OR = {
								NOT = { has_global_flag = atools_have_receiver }
								NOT = { total_development = event_target:atools_receiver }
							}
						}
						save_event_target_as = atools_receiver
						set_global_flag = atools_have_receiver
					}
				}

				if = {
					limit = { has_global_flag = atools_have_receiver }
					random_owned_province = {
						limit = {
							is_city = yes
							any_neighbor_province = { owned_by = event_target:atools_receiver }
						}
						add_core = event_target:atools_receiver
						cede_province = event_target:atools_receiver
						remove_core = event_target:atools_top_nation
						1 = { change_variable = { which = atools_cede_count value = 1 } }
					}
				}
			}
		}
	}
}

# === TEMP (region-breakaway task 1): detection CONFIRMED; now testing creation ==
# A large nation's out-of-capital-region holdings are handed to a living, independent
# nation that historically cores them (the region "returns" to its rightful owner).
# release=TAG is literal-only, so we cede via random_core_country + cede_province=
# event_target (both proven). Truly invented nations = custom tag pool (backlog).
# Detection uses capital_scope + region=event_target:<capital province> (province cmp).
# Guarded by flags (exists=event_target is unreliable). Traced; fired from the timer.
atools_split_effect = {
	clr_global_flag = atools_split_have
	clr_global_flag = atools_split_recv

	# 1) Target: a nation from the top development bracket (likely multi-region).
	if = { limit = { any_country = { is_subject = no total_development = 800 } } random_country = { limit = { is_subject = no total_development = 800 } save_event_target_as = atools_split_nation set_global_flag = atools_split_have } }
	else_if = { limit = { any_country = { is_subject = no total_development = 500 } } random_country = { limit = { is_subject = no total_development = 500 } save_event_target_as = atools_split_nation set_global_flag = atools_split_have } }
	else_if = { limit = { any_country = { is_subject = no total_development = 300 } } random_country = { limit = { is_subject = no total_development = 300 } save_event_target_as = atools_split_nation set_global_flag = atools_split_have } }
	else_if = { limit = { any_country = { is_subject = no total_development = 150 } } random_country = { limit = { is_subject = no total_development = 150 } save_event_target_as = atools_split_nation set_global_flag = atools_split_have } }

	if = {
		limit = { has_global_flag = atools_split_have }
		log = "ATOOLS split target-found"
		event_target:atools_split_nation = {
			capital_scope = { save_event_target_as = atools_split_capprov }
			# 2) Find a living, independent nation that cores an out-of-region province we own.
			random_owned_province = {
				limit = {
					is_city = yes
					NOT = { region = event_target:atools_split_capprov }
					any_core_country = { NOT = { tag = event_target:atools_split_nation } is_subject = no num_of_cities = 1 }
				}
				random_core_country = {
					limit = { NOT = { tag = event_target:atools_split_nation } is_subject = no num_of_cities = 1 }
					save_global_event_target_as = atools_split_recv_ctry
					set_global_flag = atools_split_recv
				}
			}
		}

		# 3) Cede that nation's out-of-region cores (which we own) back to it.
		if = {
			limit = { has_global_flag = atools_split_recv }
			log = "ATOOLS split receiver-found"
			event_target:atools_split_recv_ctry = {
				every_core_province = {
					limit = {
						is_city = yes
						owned_by = event_target:atools_split_nation
						NOT = { region = event_target:atools_split_capprov }
					}
					cede_province = event_target:atools_split_recv_ctry
				}
			}
			log = "ATOOLS split ceded"
		}
		else = { log = "ATOOLS split no-receiver" }
	}
}

# === Tool: Auto-convert religion & culture ==================================
# Fights religious/cultural stagnation (vanilla AI almost never converts same-group
# "heretic" provinces or foreign cultures). Each interval, every owned province whose
# religion/culture differs from its owner's has a CHANCE to flip toward the owner, so
# the map homogenises gradually rather than all at once. change_religion/change_culture
# accept a country scope; `owner` = the province's owner (religion + primary culture).
# The chance below is the per-province, per-interval percent (tune to taste).
atools_convert_effect = {
	every_province = {
		limit = { is_city = yes }
		if = {
			limit = { NOT = { religion = owner } }
			random = { chance = 20 change_religion = owner }
		}
		if = {
			limit = { NOT = { culture = owner } }
			random = { chance = 20 change_culture = owner }
		}
	}
}
