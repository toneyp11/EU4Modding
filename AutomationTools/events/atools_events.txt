##############################################################################
# AutomationTools - BACKEND (framework: the global timer / dispatcher)
# ---------------------------------------------------------------------------
# A hidden, self-refiring monthly tick on province 1 (the "hub", always exists).
# Each tick it reads the UI<->backend contract (enable flags + interval var) and
# dispatches to the enabled tools' scripted effects. Adding a tool = add another
# "if enabled ... call its effect" block here (or in a dispatch effect).
# Started once from on_startup. See docs/automationtools.md for the contract.
##############################################################################

namespace = atools

province_event = {
	id = atools.1
	title = none
	desc = none
	hidden = yes
	is_triggered_only = yes

	immediate = {
		# Keep the monthly chain alive (~30 days; drifts slightly from calendar).
		province_event = { id = atools.1 days = 30 }

		# Elapsed-month counter, kept on the hub (ROOT = province 1 here).
		change_variable = { which = atools_months value = 1 }

		# --- Tool: Auto-cede from #1 nation ---
		# Fire when enabled and the configured interval (months) has elapsed.
		if = {
			limit = {
				has_global_flag = atools_cede_enabled
				check_variable = { which = atools_cede_interval value = 1 }             # interval >= 1
				check_variable = { which = atools_months which = atools_cede_interval }  # months >= interval
			}
			set_variable = { which = atools_months value = 0 }
			atools_cede_effect = yes
		}
	}

	# Hidden events still require one option; never shown.
	option = { name = atools.1.a }
}
