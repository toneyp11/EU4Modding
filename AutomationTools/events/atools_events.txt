##############################################################################
# AutomationTools - BACKEND (framework: the global timer / dispatcher)
# ---------------------------------------------------------------------------
# A hidden, self-refiring monthly tick on province 1 (the "hub", always exists).
# Each tick it reads the UI<->backend contract (enable flags + interval var) and
# dispatches to the enabled tools' scripted effects. Adding a tool = add another
# "if enabled ... call its effect" block here (or in a dispatch effect).
# Started once from on_startup. See docs/automationtools.md for the contract.
##############################################################################

namespace = atools

province_event = {
	id = atools.1
	title = none
	desc = none
	hidden = yes
	is_triggered_only = yes

	immediate = {
		# Keep the monthly chain alive (~30 days; drifts slightly from calendar).
		province_event = { id = atools.1 days = 30 }

		# Each tool has its OWN elapsed-month counter + interval, so their cadences are
		# independent. Bump both counters every tick; when a counter reaches its tool's
		# interval, reset it and run that tool (if enabled).
		change_variable = { which = atools_months value = 1 }
		change_variable = { which = atools_convert_months value = 1 }
		change_variable = { which = atools_split_months value = 1 }

		# Anti-shrink tool (atools_cede_interval / atools_months).
		if = {
			limit = {
				check_variable = { which = atools_cede_interval value = 1 }             # interval >= 1
				check_variable = { which = atools_months which = atools_cede_interval }  # months >= interval
			}
			set_variable = { which = atools_months value = 0 }
			if = { limit = { has_global_flag = atools_cede_enabled } atools_cede_effect = yes }
		}

		# Auto-convert tool (atools_convert_interval / atools_convert_months).
		if = {
			limit = {
				check_variable = { which = atools_convert_interval value = 1 }                 # interval >= 1
				check_variable = { which = atools_convert_months which = atools_convert_interval } # months >= interval
			}
			set_variable = { which = atools_convert_months value = 0 }
			if = { limit = { has_global_flag = atools_convert_enabled } atools_convert_effect = yes }
		}

		# Region-based breakaway tool (atools_split_interval / atools_split_months).
		if = {
			limit = {
				check_variable = { which = atools_split_interval value = 1 }               # interval >= 1
				check_variable = { which = atools_split_months which = atools_split_interval } # months >= interval
			}
			set_variable = { which = atools_split_months value = 0 }
			if = { limit = { has_global_flag = atools_split_enabled } atools_split_effect = yes }
		}
	}

	# Hidden events still require one option; never shown.
	option = { name = atools.1.a }
}
